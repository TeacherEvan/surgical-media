scene_router_code = '''#!/usr/bin/env python3
"""
Intelligent Scene Backend Router & Hardware Capability Probe.
Evaluates subject dimensionality, physics/spatial cues, and telemetry requirements
to route each scene to Headless Blender 3D (bpy), Hybrid 3D+2D Motion-Tracked HUD,
Manim Math, or 2D Supersampled Canvas.
"""

import os
import re
import shutil
import subprocess
from typing import Any, Dict, Tuple


class SceneRouter:
    BLENDER_CATEGORIES = {
        "exploded_hardware_3d",
        "orbital_or_field_3d",
        "network_topology_3d",
        "rigid_body_physics_3d",
        "mathematical_manifold_3d",
        "external_asset_3d",
    }

    KEYWORDS_3D = {
        "3d", "hardware", "chip", "gpu", "die", "pcb", "reactor", "tokamak", "stellarator",
        "plasma", "orbit", "satellite", "planet", "molecule", "protein", "dna", "lattice",
        "gear", "robotics", "actuator", "drone", "chassis", "cutaway", "exploded",
        "physics", "collision", "domino", "fluid", "manifold", "topology", "volumetric",
        "server rack", "submarine cable", "fiber", "turbine", "engine", "battery cell",
    }

    KEYWORDS_MATH = {
        "equation", "integral", "derivative", "matrix", "eigenvalue", "fourier", "theorem",
        "proof", "latex", "calculus",
    }

    @staticmethod
    def probe_blender_environment() -> Dict[str, Any]:
        """Detects installed Blender binary, version (3.6 vs 4.0 vs 4.2+), and optimal EEVEE/Cycles engine enum."""
        blender_bin = os.environ.get("BLENDER_BIN") or shutil.which("blender")
        if not blender_bin:
            return {
                "available": False,
                "binary": None,
                "version": (0, 0, 0),
                "eevee_engine_id": "BLENDER_EEVEE_NEXT",
            }
        try:
            out = subprocess.check_output([blender_bin, "--version"], text=True, stderr=subprocess.DEVNULL)
            m = re.search(r"Blender\\s+(\\d+)\\.(\\d+)(?:\\.(\\d+))?", out)
            if m:
                ver = (int(m.group(1)), int(m.group(2)), int(m.group(3) or 0))
            else:
                ver = (4, 0, 0)
            # Blender 4.2+ replaced BLENDER_EEVEE with BLENDER_EEVEE_NEXT
            eevee_id = "BLENDER_EEVEE_NEXT" if (ver[0] > 4 or (ver[0] == 4 and ver[1] >= 2)) else "BLENDER_EEVEE"
            return {
                "available": True,
                "binary": blender_bin,
                "version": ver,
                "eevee_engine_id": eevee_id,
            }
        except Exception:
            return {
                "available": False,
                "binary": None,
                "version": (0, 0, 0),
                "eevee_engine_id": "BLENDER_EEVEE_NEXT",
            }

    @classmethod
    def route_scene(cls, scene: Dict[str, Any]) -> str:
        """
        Determines optimal backend ('blender_hybrid', 'blender_3d', 'manim_math', or 'canvas_2d')
        from explicit configuration or automatic semantic + category scoring.
        """
        explicit = scene.get("backend", "auto")
        if explicit != "auto":
            return explicit

        spec = scene.get("scene_spec", {})
        cat = spec.get("category", "")
        narration = (scene.get("narration_text", "") + " " + str(spec.get("params", ""))).lower()
        anchors = spec.get("anchors_3d", [])

        if cat in cls.BLENDER_CATEGORIES:
            return "blender_hybrid" if len(anchors) > 0 else "blender_3d"

        score_3d = sum(1 for kw in cls.KEYWORDS_3D if kw in narration)
        score_math = sum(1 for kw in cls.KEYWORDS_MATH if kw in narration)

        if score_3d >= 1 or spec.get("boolean_cutaway") or spec.get("explode_factor", 0) > 0:
            return "blender_hybrid" if len(anchors) > 0 else "blender_3d"
        if score_math >= 2:
            return "manim_math"
        return "canvas_2d"
'''

def hex_to_linear_rgba(hex_str: str, alpha: float = 1.0) -> Tuple[float, float, float, float]:
    h = hex_str.lstrip("#")
    if len(h) == 3:
        h = "".join(c * 2 for c in h)
    srgb = [int(h[i : i + 2], 16) / 255.0 for i in (0, 2, 4)]
    # sRGB to Linear approximation for physically based Blender shaders
    lin = [c / 12.92 if c <= 0.04045 else math.pow((c + 0.055) / 1.055, 2.4) for c in srgb]
    return (lin[0], lin[1], lin[2], alpha)

def set_bsdf_socket(bsdf_node, socket_candidates: List[str], value: Any):
    """Safely assigns a Principled BSDF input across Blender 3.6 LTS and Blender 4.0+/4.2+."""
    for name in socket_candidates:
        if name in bsdf_node.inputs:
            bsdf_node.inputs[name].default_value = value
            return True
    return False


def create_pbr_material(
    name: str,
    base_hex: str = "#1E293B",
    metallic: float = 0.85,
    roughness: float = 0.22,
    emission_hex: str = "#000000",
    emission_strength: float = 0.0,
    transmission: float = 0.0,
    coat: float = 0.0,
) -> "bpy.types.Material":
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF") or nodes.new(type="ShaderNodeBsdfPrincipled")

    set_bsdf_socket(bsdf, ["Base Color"], hex_to_linear_rgba(base_hex, 1.0))
    set_bsdf_socket(bsdf, ["Metallic"], float(metallic))
    set_bsdf_socket(bsdf, ["Roughness"], float(roughness))
    set_bsdf_socket(bsdf, ["Coat Weight", "Clearcoat"], float(coat))
    set_bsdf_socket(bsdf, ["Transmission Weight", "Transmission"], float(transmission))

    if emission_strength > 0.0:
        set_bsdf_socket(bsdf, ["Emission Color", "Emission"], hex_to_linear_rgba(emission_hex, 1.0))
        set_bsdf_socket(bsdf, ["Emission Strength"], float(emission_strength))

    return mat

def configure_render_pipeline(
    scene: "bpy.types.Scene",
    width: int,
    height: int,
    fps: int,
    total_frames: int,
    engine_pref: str = "AUTO_EEVEE",
    transparent_bg: bool = False,
):
    scene.render.resolution_x = int(width)
    scene.render.resolution_y = int(height)
    scene.render.resolution_percentage = 100
    scene.render.fps = int(fps)
    scene.frame_start = 1
    scene.frame_end = int(total_frames)
    scene.render.film_transparent = bool(transparent_bg)
    scene.render.use_motion_blur = True
    scene.render.motion_blur_shutter = 0.45

    # Select engine compatible with installed Blender version
    ver = bpy.app.version
    if engine_pref == "CYCLES":
        scene.render.engine = "CYCLES"
        scene.cycles.samples = 64
        scene.cycles.use_adaptive_sampling = True
        scene.cycles.use_denoising = True
        # Attempt GPU acceleration
        try:
            prefs = bpy.context.preferences.addons["cycles"].preferences
            for dev_type in ("OPTIX", "CUDA", "HIP", "METAL"):
                try:
                    prefs.compute_device_type = dev_type
                    prefs.get_devices()
                    if any(d.type == dev_type for d in prefs.devices):
                        for d in prefs.devices:
                            d.use = True
                        scene.cycles.device = "GPU"
                        break
                except Exception:
                    continue
        except Exception:
            pass
    else:
        scene.render.engine = "BLENDER_EEVEE_NEXT" if (ver[0] > 4 or (ver[0] == 4 and ver[1] >= 2)) else "BLENDER_EEVEE"
        if hasattr(scene, "eevee"):
            if hasattr(scene.eevee, "taa_render_samples"):
                scene.eevee.taa_render_samples = 64
            if hasattr(scene.eevee, "use_bloom"):
                scene.eevee.use_bloom = True
            if hasattr(scene.eevee, "use_ssr"):
                scene.eevee.use_ssr = True

    # Color Management: Prefer AgX -> Filmic -> Standard
    for transform in ("AgX", "Filmic", "Standard"):
        try:
            scene.view_settings.view_transform = transform
            break
        except Exception:
            continue
    for look in ("AgX - High Contrast", "High Contrast", "Medium High Contrast", "None"):
        try:
            scene.view_settings.look = look
            break
        except Exception:
            continue

    # Compositor Glare (FOG_GLOW) Node for Blender 4.2+ EEVEE Next & Cycles Bloom
    scene.use_nodes = True
    tree = scene.node_tree
    tree.nodes.clear()
    rl_node = tree.nodes.new(type="CompositorNodeRLayers")
    glare_node = tree.nodes.new(type="CompositorNodeGlare")
    glare_node.glare_type = "FOG_GLOW"
    glare_node.quality = "HIGH"
    glare_node.threshold = 0.85
    glare_node.size = 7
    comp_node = tree.nodes.new(type="CompositorNodeComposite")

    tree.links.new(rl_node.outputs["Image"], glare_node.inputs["Image"])
    tree.links.new(glare_node.outputs["Image"], comp_node.inputs["Image"])

def setup_camera_rig(scene: "bpy.types.Scene", cam_spec: Dict[str, Any], total_frames: int) -> "bpy.types.Object":
    focus_empty = bpy.data.objects.new("Focus_Target_Empty", None)
    focus_empty.location = tuple(cam_spec.get("target_pos", [0.0, 0.0, 0.0]))
    scene.collection.objects.link(focus_empty)

    cam_data = bpy.data.cameras.new("MainCamera")
    cam_data.lens = float(cam_spec.get("focal_length_mm", 50.0))
    cam_data.dof.use_dof = True
    cam_data.dof.focus_object = focus_empty
    cam_data.dof.aperture_fstop = float(cam_spec.get("fstop", 2.8))

    cam_obj = bpy.data.objects.new("MainCamera", cam_data)
    scene.collection.objects.link(cam_obj)
    scene.camera = cam_obj

    # Track-To Constraint locked to Focus_Target_Empty
    track = cam_obj.constraints.new(type="TRACK_TO")
    track.target = focus_empty
    track.track_axis = "TRACK_NEGATIVE_Z"
    track.up_axis = "UP_Y"

    r0 = float(cam_spec.get("radius_start", 6.2))
    r1 = float(cam_spec.get("radius_end", 4.4))
    az0 = math.radians(float(cam_spec.get("azimuth_start_deg", -40.0)))
    az1 = math.radians(float(cam_spec.get("azimuth_end_deg", 40.0)))
    elev = math.radians(float(cam_spec.get("elevation_deg", 24.0)))

    for f in range(1, total_frames + 1):
        u = (f - 1) / float(max(1, total_frames - 1))
        # Smooth cubic ease-in-out
        u_ease = 4.0 * u * u * u if u < 0.5 else 1.0 - math.pow(-2.0 * u + 2.0, 3.0) / 2.0
        r = r0 + (r1 - r0) * u_ease
        az = az0 + (az1 - az0) * u_ease
        cx = r * math.cos(elev) * math.sin(az)
        cy = -r * math.cos(elev) * math.cos(az)
        cz = r * math.sin(elev)
        cam_obj.location = (cx, cy, cz)
        cam_obj.keyframe_insert(data_path="location", frame=f)

    return cam_obj

def build_exploded_hardware_3d(scene: "bpy.types.Scene", spec: Dict[str, Any], total_frames: int) -> Dict[str, "bpy.types.Object"]:
    """Builds a 4-layer 3D semiconductor/hardware stack with animated vertical explosion & optional boolean cutaway."""
    mat_pcb = create_pbr_material("PCB_Substrate", "#0F172A", metallic=0.3, roughness=0.4)
    mat_gold = create_pbr_material("Gold_Contacts", "#FACC15", metallic=0.95, roughness=0.15, emission_hex="#FACC15", emission_strength=0.4)
    mat_die = create_pbr_material("Silicon_Tensor_Die", "#38BDF8", metallic=0.9, roughness=0.1, emission_hex="#38BDF8", emission_strength=2.2, coat=1.0)
    mat_heatsink = create_pbr_material("Anodized_Heatsink", "#334155", metallic=0.88, roughness=0.25)

    layers = [
        ("anchor_substrate", "Package Substrate", (0.0, 0.0, -0.65), (2.2, 2.2, 0.14), mat_pcb, -0.55),
        ("anchor_interposer", "CoWoS HBM3e Interposer", (0.0, 0.0, -0.20), (1.65, 1.65, 0.09), mat_gold, -0.15),
        ("anchor_compute_die", "5nm AI Compute Die", (0.0, 0.0, 0.20), (1.15, 1.15, 0.12), mat_die, 0.25),
        ("anchor_vapor_chamber", "3D Vapor Chamber Fins", (0.0, 0.0, 0.70), (2.0, 2.0, 0.22), mat_heatsink, 0.75),
    ]

    anchor_map: Dict[str, "bpy.types.Object"] = {}
    explode_amp = float(spec.get("explode_factor", 1.0))

    for aid, name, base_pos, scale_xyz, mat, exp_dz in layers:
        bpy.ops.mesh.primitive_cube_add(size=1.0, location=base_pos)
        obj = bpy.context.active_object
        obj.name = name
        obj.scale = scale_xyz
        bpy.ops.object.modifier_add(type="BEVEL")
        obj.modifiers["Bevel"].width = 0.03
        obj.modifiers["Bevel"].segments = 3
        obj.data.materials.append(mat)

        for f in range(1, total_frames + 1):
            u = (f - 1) / float(max(1, total_frames - 1))
            pulse = math.sin(u * math.pi) * explode_amp
            obj.location = (base_pos[0], base_pos[1], base_pos[2] + exp_dz * pulse)
            obj.keyframe_insert(data_path="location", frame=f)

        anchor_map[aid] = obj

    return anchor_map


def build_orbital_or_field_3d(scene: "bpy.types.Scene", spec: Dict[str, Any], total_frames: int) -> Dict[str, "bpy.types.Object"]:
    """Builds a 3D Toroidal Magnetic Confinement / Stellarator Cage with Twisted Field Coils & Plasma Core."""
    mat_plasma = create_pbr_material("Plasma_Core", "#38BDF8", metallic=0.1, roughness=0.1, emission_hex="#38BDF8", emission_strength=4.5)
    mat_coil = create_pbr_material("Superconducting_Coil", "#F59E0B", metallic=0.92, roughness=0.18, emission_hex="#F59E0B", emission_strength=0.8)
    mat_vessel = create_pbr_material("Vacuum_Vessel", "#475569", metallic=0.85, roughness=0.30)

    plasma.name = "Toroidal_Plasma_Core"
    plasma.data.materials.append(mat_plasma)

    coil_parent = bpy.data.objects.new("Magnetic_Coils_Rig", None)
    scene.collection.objects.link(coil_parent)
    num_coils = 12
    for i in range(num_coils):
        ang = (2.0 * math.pi * i) / num_coils
        cx = 1.65 * math.cos(ang)
        cy = 1.65 * math.sin(ang)
        bpy.ops.mesh.primitive_torus_add(major_radius=0.62, minor_radius=0.055, location=(cx, cy, 0.0))
        c_obj = bpy.context.active_object
        c_obj.rotation_euler = Euler((math.radians(90.0), math.sin(ang * 3.0) * 0.35, ang + math.pi / 2.0), "XYZ")
        c_obj.data.materials.append(mat_coil)
        c_obj.parent = coil_parent

    for f in range(1, total_frames + 1):
        u = (f - 1) / float(max(1, total_frames - 1))
        plasma.rotation_euler = Euler((0.0, 0.0, u * math.pi * 2.0), "XYZ")
        plasma.keyframe_insert(data_path="rotation_euler", frame=f)

    # Create named 3D anchor empties for 2D HUD tracking
    anchors = {}
    for a_spec in spec.get("anchors_3d", []):

        emp.location = tuple(a_spec["local_pos"])

def export_anchor_tracks(
    scene: "bpy.types.Scene",
    cam_obj: "bpy.types.Object",
    spec: Dict[str, Any],
    anchor_objs: Dict[str, "bpy.types.Object"],
    total_frames: int,
    out_json_path: str,
):
    """Computes exact 2D screen-space coordinates (u, v, z_depth) per frame using world_to_camera_view."""
    tracks_per_frame: List[List[Dict[str, Any]]] = []
    anchor_defs = spec.get("anchors_3d", [])

    for f in range(1, total_frames + 1):
        scene.frame_set(f)
        bpy.context.view_layer.update()
        frame_anchors = []
        for a_def in anchor_defs:
            aid = a_def["anchor_id"]
            if aid in anchor_objs:
                world_pos = anchor_objs[aid].matrix_world.translation
            else:
                world_pos = Vector(a_def["local_pos"])
            co_2d = world_to_camera_view(scene, cam_obj, world_pos)
            # Note: Blender world_to_camera_view has origin at bottom-left; convert v to top-left [0..1]
            u_screen = float(co_2d.x)
            v_screen = float(1.0 - co_2d.y)
            z_depth = float(co_2d.z)
            frame_anchors.append(
                {
                    "anchor_id": aid,
                    "label": a_def["label"],
                    "metric": a_def["metric"],
                    "accent_hex": a_def.get("accent_hex", "#38BDF8"),
                    "u": round(u_screen, 5),
                    "v": round(v_screen, 5),
                    "z_depth": round(z_depth, 4),
                    "visible": bool(z_depth > 0.0 and 0.02 <= u_screen <= 0.98 and 0.02 <= v_screen <= 0.98),
                }
            )
        tracks_per_frame.append(frame_anchors)

    with open(out_json_path, "w", encoding="utf-8") as fp:
        json.dump({"total_frames": total_frames, "frames": tracks_per_frame}, fp, indent=2)

software_3d_code = '''#!/usr/bin/env python3
"""
True 4x4 Perspective Matrix 3D Mesh Shader & Screen-Space Projection Tracker.
Provides an exact mathematical mirror of Blender's Track-To Camera Rig and
`bpy_extras.object_utils.world_to_camera_view` for environments without a local
Blender binary or for rapid pre-visualization of 3D scenes + 2D HUD anchor tracks.
"""

import math
from typing import Any, Dict, List, Tuple
import numpy as np
from PIL import Image, ImageDraw, ImageFilter
from camera_matrix import AffineSupersampledCanvas, hex_to_rgba


class PerspectiveCamera3D:
    """Computes Look-At View Matrix V and Perspective Projection Matrix P matching Blender's sensor model."""

    def __init__(
        self,
        eye: Tuple[float, float, float],
        target: Tuple[float, float, float] = (0.0, 0.0, 0.0),
        up: Tuple[float, float, float] = (0.0, 0.0, 1.0),
        focal_length_mm: float = 50.0,
        sensor_width_mm: float = 36.0,
        aspect_ratio: float = 16.0 / 9.0,
    ):
        self.eye = np.array(eye, dtype=np.float64)
        self.target = np.array(target, dtype=np.float64)
        self.up = np.array(up, dtype=np.float64)
        self.aspect = float(aspect_ratio)
        self.fov_x = 2.0 * math.atan(sensor_width_mm / (2.0 * focal_length_mm))
        self.f_scale = 1.0 / math.tan(self.fov_x * 0.5)

        forward = self.target - self.eye
        f_norm = np.linalg.norm(forward)
        self.forward = forward / (f_norm if f_norm > 1e-8 else 1.0)

        right = np.cross(self.forward, self.up)
        r_norm = np.linalg.norm(right)
        self.right = right / (r_norm if r_norm > 1e-8 else 1.0)
        self.true_up = np.cross(self.right, self.forward)

    def project_point(self, pt_world: Tuple[float, float, float]) -> Tuple[float, float, float, bool]:
        """
        Mirrors bpy_extras.object_utils.world_to_camera_view:
        Returns normalized screen coordinates (u in [0..1] left-to-right, v in [0..1] top-to-bottom, z_depth, visible).
        """
        p = np.array(pt_world, dtype=np.float64) - self.eye
        x_cam = float(np.dot(p, self.right))
        y_cam = float(np.dot(p, self.true_up))
        z_cam = float(np.dot(p, self.forward))

        if z_cam <= 0.15:
            return (0.5, 0.5, z_cam, False)

        ndc_x = (x_cam / z_cam) * self.f_scale
        ndc_y = (y_cam / z_cam) * self.f_scale * self.aspect

        u = 0.5 + 0.5 * ndc_x
        v = 0.5 - 0.5 * ndc_y
        vis = (0.02 <= u <= 0.98) and (0.02 <= v <= 0.98)
        return (u, v, z_cam, vis)

def generate_torus_stellarator_mesh(t_norm: float) -> List[Dict[str, Any]]:
    """Generates a 3D Toroidal Plasma & Twisted Modular Stellarator Coil Mesh."""
    polys = []
    R = 1.65
    r = 0.36
    n_maj = 32
    n_min = 14
    rot_z = t_norm * math.pi * 1.5

    for i in range(n_maj):
        u0 = (2.0 * math.pi * i) / n_maj + rot_z
        u1 = (2.0 * math.pi * (i + 1)) / n_maj + rot_z
        is_coil = (i % 3 == 0)
        for j in range(n_min):
            v0 = (2.0 * math.pi * j) / n_min
            v1 = (2.0 * math.pi * (j + 1)) / n_min
            r_eff = r * (1.28 if is_coil else 1.0)
            twist = 0.25 * math.sin(3.0 * u0) if is_coil else 0.0

            def pt(u, v):
                x = (R + r_eff * math.cos(v + twist)) * math.cos(u)
                y = (R + r_eff * math.cos(v + twist)) * math.sin(u)
                z = r_eff * math.sin(v + twist)
                return (x, y, z)

            p00, p10, p11, p01 = pt(u0, v0), pt(u1, v0), pt(u1, v1), pt(u0, v1)
            nx = math.cos(v0) * math.cos(u0)
            ny = math.cos(v0) * math.sin(u0)
            nz = math.sin(v0)
            polys.append(
                {
                    "verts": [p00, p10, p11, p01],
                    "normal": (nx, ny, nz),
                    "color_hex": "#F59E0B" if is_coil else "#0284C7",
                    "edge_hex": "#FDE047" if is_coil else "#38BDF8",
                    "emissive": is_coil or (j % 4 == 0),
                }
            )
    return polys


def render_3d_scene_and_track(
    canvas: AffineSupersampledCanvas,
    scene_spec: Dict[str, Any],
    t_norm: float,
) -> List[Dict[str, Any]]:
    """
    Renders the 3D scene using 4x4 perspective projection + Painter's Z-buffer + Blinn-Phong lighting
    and returns the exact screen-space projected `anchors_3d` list for the 2D HUD compositor.
    """
    w, h = canvas.target_w, canvas.target_h
    cam_cfg = scene_spec.get("camera_3d", {})
    r0 = float(cam_cfg.get("radius_start", 6.2))
    r1 = float(cam_cfg.get("radius_end", 4.5))
    az0 = math.radians(float(cam_cfg.get("azimuth_start_deg", -38.0)))
    az1 = math.radians(float(cam_cfg.get("azimuth_end_deg", 42.0)))
    elev = math.radians(float(cam_cfg.get("elevation_deg", 25.0)))

    u_ease = 4.0 * (t_norm ** 3) if t_norm < 0.5 else 1.0 - math.pow(-2.0 * t_norm + 2.0, 3.0) / 2.0
    radius = r0 + (r1 - r0) * u_ease
    az = az0 + (az1 - az0) * u_ease

    eye = (
        radius * math.cos(elev) * math.sin(az),
        -radius * math.cos(elev) * math.cos(az),
        radius * math.sin(elev),
    )
    cam3d = PerspectiveCamera3D(
        eye=eye,
        target=(0.0, 0.0, 0.0),
        focal_length_mm=float(cam_cfg.get("focal_length_mm", 50.0)),
        aspect_ratio=float(w) / float(h),
    )

    # 1. Draw 3D Perspective Ground Floor Grid (z = -1.25)
    grid_z = -1.25
    for g in range(-3, 4):
        u0, v0, _, vis0 = cam3d.project_point((g * 0.9, -2.7, grid_z))
        u1, v1, _, vis1 = cam3d.project_point((g * 0.9, 2.7, grid_z))
        if vis0 and vis1:
            canvas.line(u0 * w, v0 * h, u1 * w, v1 * h, fill=(56, 189, 248, 32), width=1.4, hud_space=True)
        u2, v2, _, vis2 = cam3d.project_point((-2.7, g * 0.9, grid_z))
        u3, v3, _, vis3 = cam3d.project_point((2.7, g * 0.9, grid_z))
        if vis2 and vis3:
            canvas.line(u2 * w, v2 * h, u3 * w, v3 * h, fill=(56, 189, 248, 32), width=1.4, hud_space=True)

    # 2. Build 3D Polygon Soup & Compute Dynamic Anchor World Positions
    cat = scene_spec.get("category", "exploded_hardware_3d")
    polys: List[Dict[str, Any]] = []
    dynamic_anchor_pos: Dict[str, Tuple[float, float, float]] = {}

    if cat == "orbital_or_field_3d":
        polys = generate_torus_stellarator_mesh(t_norm)
    else:
        explode_amp = float(scene_spec.get("explode_factor", 1.0)) * math.sin(min(1.0, t_norm * 1.15) * math.pi * 0.65)
        layers = [
            ("anchor_substrate", (0.0, 0.0, -0.65 - 0.50 * explode_amp), (2.4, 2.4, 0.16), "#0F172A", "#38BDF8", False),
            ("anchor_interposer", (0.0, 0.0, -0.20 - 0.15 * explode_amp), (1.85, 1.85, 0.11), "#78350F", "#FACC15", True),
            ("anchor_compute_die", (0.0, 0.0, 0.22 + 0.25 * explode_amp), (1.28, 1.28, 0.15), "#0369A1", "#38BDF8", True),
            ("anchor_vapor_chamber", (0.0, 0.0, 0.72 + 0.70 * explode_amp), (2.15, 2.15, 0.22), "#1E293B", "#94A3B8", False),
        ]
        for aid, pos, scl, c_hex, e_hex, emis in layers:
            polys.extend(generate_box_mesh(pos, scl, c_hex, e_hex, emissive=emis))
            dynamic_anchor_pos[aid] = (pos[0] + scl[0] * 0.42, pos[1] - scl[1] * 0.42, pos[2])

    # 3. Depth-Sort Polygons (Painter's Algorithm) & Shade with Blinn-Phong Lighting
    light_dir = np.array([0.55, -0.45, 0.70], dtype=np.float64)
    light_dir /= np.linalg.norm(light_dir)

    projected_polys = []
    for poly in polys:
        pts_2d = []
        depths = []
        all_vis = True
        for vx, vy, vz in poly["verts"]:
            u, v, z_d, vis = cam3d.project_point((vx, vy, vz))
            if z_d <= 0.2:
                all_vis = False
                break
            pts_2d.append((int(round(u * w * canvas.scale)), int(round(v * h * canvas.scale))))
            depths.append(z_d)
        if all_vis and pts_2d:
            avg_z = sum(depths) / len(depths)
            projected_polys.append((avg_z, pts_2d, poly))

    projected_polys.sort(key=lambda item: item[0], reverse=True)

    for _, pts_2d, poly in projected_polys:
        norm = np.array(poly["normal"], dtype=np.float64)
        ndotl = max(0.0, float(np.dot(norm, light_dir)))
        intensity = 0.38 + 0.62 * ndotl
        base_r, base_g, base_b, _ = hex_to_rgba(poly["color_hex"], 235)
        shaded_rgba = (
            min(255, int(base_r * intensity)),
            min(255, int(base_g * intensity)),
            min(255, int(base_b * intensity)),

        )
        edge_rgba = hex_to_rgba(poly["edge_hex"], 240)
        canvas.base_draw.polygon(pts_2d, fill=shaded_rgba, outline=edge_rgba)
        if poly.get("emissive"):
            canvas._has_bloom = True
            canvas.bloom_draw.polygon(pts_2d, fill=None, outline=edge_rgba)

    # 4. Project Tagged 3D Anchors into Normalized 2D Screen Space (u, v, z_depth)
    tracked_anchors = []

        aid = a_def["anchor_id"]

                "label": a_def["label"],
                "metric": a_def["metric"],
                "accent_hex": a_def.get("accent_hex", "#38BDF8"),
                "u": u,
                "v": v,
                "z_depth": z_d,
                "visible": vis,
            }
        )

    return tracked_anchors

def render_word_synced_captions(
    canvas: AffineSupersampledCanvas,
    word_timings: List[Dict[str, Any]],
    scene_time_sec: float,
    aspect_ratio: str = "16:9",
):
    if not word_timings:
        return
    w, h = canvas.target_w, canvas.target_h
    active_idx = 0
    for idx, wt in enumerate(word_timings):
        if scene_time_sec >= float(wt["start_sec"]):
            active_idx = idx

    chunk_size = 6 if aspect_ratio == "16:9" else 4
    chunk_start = (active_idx // chunk_size) * chunk_size
    chunk = word_timings[chunk_start : chunk_start + chunk_size]

    font_cap = canvas.get_font(30 if aspect_ratio == "16:9" else 36, mono=False, hud_space=True)
    y_pos = h * 0.865 if aspect_ratio == "16:9" else h * 0.77

    full_str = " ".join(item["word"] for item in chunk)
    bbox = canvas.base_draw.textbbox((0, 0), full_str, font=font_cap)
    line_w = (bbox[2] - bbox[0]) / float(canvas.scale)
    curr_x = (w - line_w) * 0.5

    canvas.glass_card(
        curr_x - 26,
        y_pos - 12,
        curr_x + line_w + 26,
        y_pos + 48,
        radius=12,
        tint_rgba=(10, 14, 23, 225),
        border_rgba=(255, 255, 255, 75),
        hud_space=True,
    )

    for i, item in enumerate(chunk):
        global_i = chunk_start + i
        word_str = item["word"]
        is_active = global_i == active_idx
        col = (255, 215, 0, 255) if is_active else (240, 244, 250, 235)
        canvas.text(curr_x, y_pos, word_str, font_cap, fill=col, emissive=is_active, hud_space=True)
        wb = canvas.base_draw.textbbox((0, 0), word_str + " ", font=font_cap)
        curr_x += (wb[2] - wb[0]) / float(canvas.scale)


def render_single_frame(
    sc: Dict[str, Any],
    f_idx: int,
    n_frames: int,
    w: int,
    h: int,
    fps: int,
    ss_scale: int,
    aspect: str,
    blender_env: Dict[str, Any],
    blender_cache_dir: str = "",
) -> Image.Image:
    t_norm = f_idx / float(max(1, n_frames - 1))
    t_sec = f_idx / float(fps)
    dur = float(sc["duration_sec"])
    narration = sc.get("narration_text", "")
    word_timings = sc.get("word_timings") or compute_word_timings(narration, dur)

    routed_backend = SceneRouter.route_scene(sc)
    spec = sc.get("scene_spec", {})
    cat = spec.get("category", "exploded_hardware_3d")

    canvas = AffineSupersampledCanvas(w, h, scale=ss_scale, bg_color="#050811")

    if routed_backend in ("blender_hybrid", "blender_3d"):
        # If pre-rendered headless Blender PNGs + anchor_tracks.json exist in blender_cache_dir, composite over them
        png_path = os.path.join(blender_cache_dir, f"frame_{f_idx + 1:04d}.png")
        tracks_path = os.path.join(blender_cache_dir, "anchor_tracks.json")
        if blender_cache_dir and os.path.exists(png_path) and os.path.exists(tracks_path):
            b_img = Image.open(png_path).convert("RGBA").resize((w * ss_scale, h * ss_scale), Image.Resampling.LANCZOS)
            canvas.base_img.paste(b_img, (0, 0), b_img)
            with open(tracks_path, "r", encoding="utf-8") as tf:
                tracks_data = json.load(tf)
            tracked_anchors = tracks_data["frames"][min(f_idx, len(tracks_data["frames"]) - 1)]
        else:
            # Execute mathematical 4x4 Perspective Matrix 3D Shader + Exact Screen-Space Tracker
            tracked_anchors = render_3d_scene_and_track(canvas, spec, t_norm)

        if routed_backend == "blender_hybrid":
            title_str = spec.get("hud_title", f"3D HYBRID // {cat.upper()}")
            composite_3d_tracked_hud(canvas, tracked_anchors, title_str, t_norm, aspect_ratio=aspect)
    else:
        # 2D Canvas Backends
        if cat == "procedural_2d_diagram":
            render_geometry_scene(canvas, spec, t_norm)
        elif cat == "data_chart_2d":
            render_doc_essay_scene(canvas, spec, t_norm)
        else:
            render_code_report_scene(canvas, spec, t_norm)

    render_word_synced_captions(canvas, word_timings, t_sec, aspect_ratio=aspect)
    return canvas.resolve()


def execute_v3_pipeline(storyboard: Dict[str, Any], output_mp4: str, preview_dir: str = "") -> Dict[str, Any]:
    meta = storyboard["meta"]
    w = int(meta["width"])
    h = int(meta["height"])
    fps = int(meta.get("fps", 60))
    aspect = meta.get("aspect_ratio", "16:9")
    ss_scale = int(meta.get("supersample_2d", 2))

    blender_env = SceneRouter.probe_blender_environment()
    scenes = storyboard["scenes"]
    total_dur = sum(float(s["duration_sec"]) for s in scenes)

    if preview_dir:
        os.makedirs(preview_dir, exist_ok=True)

    # Synthesize 48 kHz Stereo Audio Track
    audio_cfg = storyboard.get("audio_master", {})
    sr = int(audio_cfg.get("sample_rate", 48000))
    mixer = StereoDSPMixer(total_duration_sec=total_dur, sample_rate=sr)
    mixer.synthesize_ambient_pad(base_freq_hz=float(audio_cfg.get("ambient_pad_hz", 110.0)), gain=0.13)

    t_cursor = 0.0
    for sc in scenes:
        for ev in sc.get("sfx_events", []):
            mixer.add_sfx_event(
                time_sec=t_cursor + float(ev["time_offset_sec"]),
                sfx_type=ev["sfx_type"],
                freq_hz=float(ev.get("freq_hz", 528.0)),
                pan=float(ev.get("pan", 0.0)),
            )
        t_cursor += float(sc["duration_sec"])

    with tempfile.TemporaryDirectory() as tmpdir:
        master_wav = os.path.join(tmpdir, "master_48k.wav")
        mixer.export_master_wav(master_wav, sidechain_duck_db=float(audio_cfg.get("sidechain_duck_db", -12.0)))
        if preview_dir:
            shutil.copy2(master_wav, os.path.join(preview_dir, "master_48k.wav"))

        # Check if headless Blender is available to pre-render 3D passes
        scene_Valid_dirs: Dict[str, str] = {}
        if blender_env["available"]:
            runner_script = str(Path(__file__).resolve().parent / "blender_runner.py")
            for sc in scenes:
                if SceneRouter.route_scene(sc) in ("blender_hybrid", "blender_3d"):
                    sid = sc["scene_id"]
                    s_dir = os.path.join(tmpdir, sid)
                    os.makedirs(s_dir, exist_ok=True)
                    spec_file = os.path.join(s_dir, "spec.json")
                    with open(spec_file, "w", encoding="utf-8") as sf:
                        json.dump(sc["scene_spec"], sf)
                    n_f = max(1, int(round(float(sc["duration_sec"]) * fps)))
                    subprocess.run(
                        [
                            blender_env["binary"], "-b", "-P", runner_script, "--",
                            "--spec", spec_file, "--out-dir", s_dir,
                            "--width", str(w), "--height", str(h),
                            "--fps", str(fps), "--frames", str(n_f),
                        ],
                        check=True,
                    )
                    scene_Valid_dirs[sid] = s_dir

        has_ffmpeg = shutil.which("ffmpeg") is not None
        if has_ffmpeg:
            ffmpeg_cmd = [
                "ffmpeg", "-y", "-v", "error",
                "-f", "rawvideo", "-vcodec", "rawvideo",
                "-s", f"{w}x{h}", "-pix_fmt", "rgba", "-r", str(fps),
                "-i", "-",
                "-i", master_wav,
                "-c:v", "libx264", "-preset", "medium", "-crf", "17",
                "-profile:v", "high", "-level", "4.2", "-pix_fmt", "yuv420p",
                "-c:a", "aac", "-b:a", "192k", "-ar", str(sr),
                "-movflags", "+faststart", "-shortest", output_mp4,
            ]
            proc = subprocess.Popen(ffmpeg_cmd, stdin=subprocess.PIPE, stderr=subprocess.PIPE)
            assert proc.stdin is not None
            for sc in scenes:
                n_frames = max(1, int(round(float(sc["duration_sec"]) * fps)))
                b_dir = scene_Valid_dirs.get(sc["scene_id"], "")
                for f_idx in range(n_frames):
                    img = render_single_frame(sc, f_idx, n_frames, w, h, fps, ss_scale, aspect, blender_env, b_dir)
                    if preview_dir and f_idx == n_frames // 2:
                        img.save(os.path.join(preview_dir, f"{sc['scene_id']}_keyframe.png"))
                    proc.stdin.write(img.tobytes())
            proc.stdin.close()
            proc.wait()
            return verify_mp4(output_mp4, w, h, fps, total_dur)
        else:
            # OpenCV MP4 + Keyframe Proof Export for container environments without ffmpeg CLI
            import cv2
            import numpy as np
            fourcc = cv2.VideoWriter_fourcc(*"mp4v")

            return {
                "status": "PASSED_CONTAINER_FALLBACK",
                "blender_binary_detected": blender_env["available"],
                "output_mp4": output_mp4,
                "frames_rendered": rendered_count,
                "resolution": f"{w}x{h}",
                "preview_dir": preview_dir,

# Automated Multi-Engine Video Production Pipeline (v3.0)

Transform any user subject prompt into a broadcast-verified social media or explainer MP4 (`16:9` YouTube, `9:16` Shorts/Reels/TikTok, or `1:1` Feed at `60 fps`). Rather than locking into fixed 2D templates, this pipeline dynamically routes each scene to **Headless Blender 3D (`bpy`)**, **Hybrid 3D + Motion-Tracked 2D HUD Overlays**, **Mathematical Morphs (`Manim`)**, or **2D Procedural Canvas (`Pillow` + `NumPy`)**.

## Stage 1: Subject Investigation & Scene Backend Routing (`src/scene_router.py`)

1. **Deconstruct the Subject:** Extract 3 to 7 verifiable technical mechanisms, spatial structures, quantitative curves, or causal chains from the user's topic.
2. **Dynamic Scene Backend Selection (`SceneRouter.route_scene`):**
   - **`blender_hybrid` (Headless Blender 3D Pass + 2D Motion-Tracked HUD Pass):** Default whenever a scene involves 3D hardware/semiconductors, mechanical/robotics assemblies, astrophysical/orbital mechanics, magnetic/fluid fields, molecular lattices, architectural cutaways, or 3D spatial topologies that require pinned telemetry labels.
   - **`blender_3d` (Pure 3D Cinematic Pass):** Default for full-frame 3D physical simulations (Bullet rigid-body cascades), environmental sweeps, or dramatic lighting reveals without HUD callouts.
   - **`canvas_2d` (Affine 2x-Supersampled 2D Vector/UI Pass):** Default for syntax-highlighted IDE diffs, 2D packet state machines, scaling charts, and articulated 2D stick-figure geometry rigs.
   - **`manim_math`:** Default for symbolic LaTeX equations and formal calculus derivations.

## Stage 2: Two-Pass 3D-to-2D Motion-Tracked HUD Architecture (`world_to_camera_view`)

Never render small explanatory text as 3D extruded meshes inside perspective space where Depth of Field bokeh, motion blur, or geometry occlusion destroys legibility on mobile screens. Instead, execute the **Two-Pass Hybrid Pipeline**:

1. **Pass 1 — Headless Blender 3D Render + Screen-Space Anchor Export (`src/blender_runner.py`):**
   - Invoke `blender -b -P src/blender_runner.py -- --spec spec.json --out-dir build/scene_01 --width 1920 --height 1080 --fps 60 --frames 180`.
   - Automatically detects `BLENDER_EEVEE_NEXT` (Blender $\\ge 4.2$), `BLENDER_EEVEE` (Blender $3.6 - 4.1$), or `CYCLES` (with `OPTIX`/`CUDA`/`HIP`/`METAL` GPU acceleration), sets `AgX - High Contrast` color management, and wires a `CompositorNodeGlare` (`FOG_GLOW`) bloom node.
   - Parents `MainCamera` to a `Track To` (`-Z`, `Up Y`) constraint targeting `Focus_Target_Empty` with `dof.use_dof = True`.
   - On every frame $f \\in [1, N]$, projects every tagged 3D object or empty anchor into normalized 2D viewport coordinates $(u, v, z_{\\text{depth}})$ via `bpy_extras.object_utils.world_to_camera_view(scene, camera, obj.matrix_world.translation)` (flipping Blender's bottom-left $y$ origin to top-left $v = 1.0 - y$) and writes `anchor_tracks.json`.
   - **Container / No-Blender Fallback (`src/software_3d_projection.py`):** If `blender` is not installed on a lightweight container or Termux host, `PerspectiveCamera3D` executes the exact $4 \\times 4$ view-projection matrix math and Painter's Z-buffer Blinn-Phong shader to produce identical `anchor_tracks` telemetry.
2. **Pass 2 — 2D SAT-Solved HUD & Leader-Line Compositor (`src/hybrid_compositor.py`):**
   - Reads the rendered 3D frame and `anchor_tracks.json`.
   - Locks a central 3D stage repulsion zone and runs 50 iterations of the 2D Separating Axis Theorem (`SATLayoutSolver`) to position frosted glassmorphic telemetry cards around the 3D subject with zero overlap and strict adherence to `16:9` / `9:16` / `1:1` safe zones.
   - Draws `2x`-supersampled target reticles and 45-degree elbow leader lines (`3D anchor -> knee -> card dock`) with additive Gaussian bloom.

## Stage 3: Composable 3D Scene Generators (`src/blender_runner.py`)

- **`exploded_hardware_3d`:** Multi-layer 3D hardware/semiconductor/mechanical assemblies with keyframed explosion curves along outward normals, bevel modifiers, and optional animated `EXACT` boolean cutaway slicers.
- **`orbital_or_field_3d`:** 3D toroidal magnetic confinement cages (Stellarators/Tokamaks), orbital constellations, and twisted non-planar field coils with high-emission plasma cores.

- **`external_asset_3d`:** Imports `.glb`/`.gltf`/`.obj`/`.fbx` meshes, normalizes world bounding boxes to unit stage coordinates, and attaches 3D callout anchors.

# 2. Upgrade hybrid_compositor.py with dynamic text-bbox card widths and peripheral callout columns
hybrid_compositor_fixed = '''#!/usr/bin/env python3
"""
3D-to-2D Motion-Tracked HUD & Callout Compositor.
Reads 3D screen-space anchor tracks (from Blender's `world_to_camera_view` or `PerspectiveCamera3D`),
dynamically sizes cards from exact font bounding boxes, solves collision-free 2D placement via
`SATLayoutSolver`, and renders 2x-supersampled angled leader lines & frosted glass telemetry badges.
"""

from typing import Any, Dict, List
from camera_matrix import AffineSupersampledCanvas, hex_to_rgba, spring_rk4
from layout_solver import LayoutBox, SATLayoutSolver


def composite_3d_tracked_hud(
    canvas: AffineSupersampledCanvas,
    tracked_anchors: List[Dict[str, Any]],
    scene_title: str,
    t_norm: float,
    aspect_ratio: str = "16:9",
):
    w, h = canvas.target_w, canvas.target_h
    font_hud = canvas.get_font(21, mono=True, hud_space=True)
    font_label = canvas.get_font(18, mono=False, hud_space=True)
    font_metric = canvas.get_font(22, mono=True, hud_space=True)

    # Measure title width dynamically
    tb_hud = canvas.base_draw.textbbox((0, 0), scene_title, font=font_hud)
    hud_w = max(620.0, (tb_hud[2] - tb_hud[0]) / float(canvas.scale) + 52.0)

    # Lock central 3D model silhouette repulsion zone & HUD header so cards dock cleanly in left/right gutters
    boxes: List[LayoutBox] = [
        LayoutBox("hud_header", 96.0, 42.0, 96.0 + hud_w, 100.0, locked=True, padding=16.0),
        LayoutBox("center_3d_repulsor", w * 0.29, h * 0.18, w * 0.71, h * 0.78, locked=True, padding=20.0),
    ]

    visible_anchors = [a for a in tracked_anchors if a.get("visible", True)]
    card_dims: Dict[str, float] = {}

    for idx, anc in enumerate(visible_anchors):
        aid = anc["anchor_id"]
        tb_l = canvas.base_draw.textbbox((0, 0), anc["label"], font=font_label)
        tb_m = canvas.base_draw.textbbox((0, 0), anc["metric"], font=font_metric)
        lw = (tb_l[2] - tb_l[0]) / float(canvas.scale)
        mw = (tb_m[2] - tb_m[0]) / float(canvas.scale)
        card_w = max(290.0, max(lw, mw) + 44.0)
        card_h = 84.0
        card_dims[aid] = card_w

        ax = float(anc["u"]) * w
        ay = float(anc["v"]) * h
        # Alternate or assign to left/right peripheral telemetry columns based on anchor index
        put_left = (idx % 2 == 1)
        cx0 = 104.0 if put_left else (w - 104.0 - card_w)
        cy0 = max(135.0, min(h * 0.72, ay - card_h * 0.5))
        boxes.append(
            LayoutBox(
                box_id=aid,
                x0=cx0,
                y0=cy0,
                x1=cx0 + card_w,
                y1=cy0 + card_h,
                locked=False,
                padding=18.0,
            )
        )

    solved = SATLayoutSolver.solve(boxes, aspect_ratio=aspect_ratio, width=w, height=h, iterations=60)

    # Render 2x-Supersampled Target Reticles, 45-Degree Elbow Leader Lines, and Frosted Glass Cards
    for idx, anc in enumerate(visible_anchors):
        appear_t = max(0.0, min(1.0, (t_norm - idx * 0.06) * 3.5))
        if appear_t <= 0.02:
            continue
        pop = spring_rk4(appear_t)
        aid = anc["anchor_id"]
        box = solved[aid]
        ax = float(anc["u"]) * w
        ay = float(anc["v"]) * h
        acc_rgba = hex_to_rgba(anc.get("accent_hex", "#38BDF8"), 255)

        # Dock on right edge of left-column cards, or left edge of right-column cards
        dock_x = box.x1 if box.cx < w * 0.5 else box.x0
        dock_y = box.cy
        dx = dock_x - ax
        dy = dock_y - ay
        elbow_len = min(abs(dy), abs(dx) * 0.45)
        knee_x = ax + (elbow_len if dx >= 0 else -elbow_len)
        knee_y = dock_y

        # Glowing 3D Surface Target Reticle
        canvas.circle(ax, ay, 10.5 * pop, fill=None, outline=acc_rgba, width=2.2, emissive=True, hud_space=True)
        canvas.circle(ax, ay, 4.2 * pop, fill=(255, 255, 255, 255), outline=acc_rgba, width=2.0, emissive=True, hud_space=True)

        # 45-Degree Elbow Leader Line (3D Anchor -> Knee -> Card Dock)
        canvas.line(ax, ay, knee_x, knee_y, fill=acc_rgba, width=2.4, emissive=True, hud_space=True)
        canvas.line(knee_x, knee_y, dock_x, dock_y, fill=acc_rgba, width=2.4, emissive=True, hud_space=True)
        canvas.circle(dock_x, dock_y, 4.5, fill=acc_rgba, emissive=True, hud_space=True)

        # Frosted Glassmorphic Callout Card
        canvas.glass_card(
            box.x0, box.y0, box.x1, box.y1,
            radius=12.0,
            tint_rgba=(11, 17, 30, 225),
            border_rgba=acc_rgba,
            hud_space=True,
        )
        canvas.text(box.x0 + 20.0, box.y0 + 14.0, anc["label"], font_label, fill=(226, 232, 240, 255), hud_space=True)
        canvas.text(box.x0 + 20.0, box.y0 + 44.0, anc["metric"], font_metric, fill=acc_rgba, emissive=True, hud_space=True)

    # Top HUD Telemetry Bar
    canvas.glass_card(96.0, 42.0, 96.0 + hud_w, 100.0, radius=10.0, border_rgba=(56, 189, 248, 195), hud_space=True)
    canvas.text(118.0, 58.0, scene_title, font_hud, fill=(56, 189, 248, 255), hud_space=True)
'''
